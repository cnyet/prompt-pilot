## Global Constraints

- 不 import `lib/engine/schema.ts`、`lib/engine/sse.ts`（v2 遗产）；不改任何 v2 文件。
- `lib/llm/types.ts` 冻结不碰；本批不新增 stage 相关代码（Batch C 在 ledger 本地扩展）。
- 分隔符与 JSON 元数据绝不进入 `delta` 流或返回的正文；分隔符出现在 chunk 中间必须正确切分。
- 降级不抛错：标题返回 null（调用方用输入前 24 字兜底），说明为空数组，正文完整。
- notes 最多 3 条，超出截断；notes 元素必须是非空字符串，非字符串元素丢弃。

- 无感叹号自检：新文件里不出现「!」结尾的中文文案。

## Review Focus

1. **分隔符跨 chunk**：`<<<PROMPTPILOT_META>>>` 被网络分块劈开（如 `<<<PROMPTPILOT_` + `META>>>`）时不能把半个分隔符当正文下发；验证 = meta-split.test 的跨 chunk 用例。
2. **降级三态正文完整**：缺分隔符、坏 JSON、提前出现三种情况下，已发 `delta` 之和等于完整正文，元数据被整体忽略；验证 = meta-split.test 三个降级用例的 body 断言。
3. **幂等 in-flight 复用**：同一 requestId 并发两次只执行一次 executor，两个调用拿到同一结果且 `replayed` 语义正确（首个 false，并发方 true）；验证 = idempotency.test 并发用例。
4. **5 分钟结果复用与 LRU 驱逐**：完成结果在 TTL 内复用、超 TTL 重新执行；超过容量后最旧条目被驱逐且不影响新写入；验证 = idempotency.test 的 TTL 与容量用例（用注入时钟）。
5. **SSE 往返一致**：`encodeSseEventV3` 的输出经 `parseSseStreamV3` 解码后与原消息逐字段相等，且任意分块边界、CRLF、注释行都能解析；验证 = sse-v3.test 的往返与分块用例。

```ts
/**
 * v3 元数据协议（ARCHITECTURE §五）。
 *
 * 模型先输出完整 Markdown 正文，正文结束后另起一行输出
 * `<<<PROMPTPILOT_META>>>`，再输出 {"title":"...","notes":[...]}。
 * 服务端边流边切分：分隔符前实时下发 delta，分隔符后缓冲，
 * 结束时解析并发一条 meta 事件。分隔符与 JSON 绝不进入正文。
 *
 * 解析失败（缺分隔符 / 坏 JSON / 分隔符提前出现）一律降级：
 * title 为 null（调用方取输入前 24 字兜底），notes 为空数组，正文不受影响。
 */

export const META_SEPARATOR = '<<<PROMPTPILOT_META>>>'

export interface MetaResult {
  /** null 表示解析失败或未提供，调用方用输入前 24 字兜底（NFR-11） */
  title: string | null
  /** 最多 3 条改动说明；生成类通常为空数组 */
  notes: string[]
}

const MAX_NOTES = 3

/** 分隔符可能被 chunk 劈开；持有尾部不足一个完整分隔符长度的字符不放行。 */
const HOLD_BACK = META_SEPARATOR.length - 1

interface RawMeta {
  title: unknown
  notes: unknown
}

function parseMetaJson(raw: string): RawMeta | null {
  try {
    const value: unknown = JSON.parse(raw)
    if (typeof value !== 'object' || value === null) {
      return null
    }
    const record = value as Record<string, unknown>
    return { title: record.title, notes: record.notes }
  } catch {
    return null
  }
}

function sanitizeNotes(notes: unknown): string[] {
  if (!Array.isArray(notes)) {
    return []
  }
  return notes.filter((item): item is string => typeof item === 'string' && item.trim().length > 0)
}

export class MetaSplitter {
  private pending = ''
  private tail = ''
  private separatorSeen = false
  private metaRaw: string | null = null
  private bodyEmitted = 0

  /** 返回本次可以立即下发到 delta 的正文片段；元数据段永不返回。 */
  push(chunk: string): string {
    this.pending += chunk
    if (!this.separatorSeen) {
      return this.drainBeforeSeparator()
    }
    // 分隔符已出现：全部缓冲为元数据
    this.metaRaw = (this.metaRaw ?? '') + this.pending
    this.pending = ''
    return ''
  }

  finish(): MetaResult {
    if (!this.separatorSeen) {
      // 缺分隔符：降级，正文已由 push + flushTail 完整放行
      return { title: null, notes: [] }
    }
    const parsed = parseMetaJson(this.metaRaw ?? '')
    if (!parsed) {
      return { title: null, notes: [] }
    }
    const title = typeof parsed.title === 'string' && parsed.title.trim().length > 0
      ? parsed.title.trim()
      : null
    return { title, notes: sanitizeNotes(parsed.notes).slice(0, MAX_NOTES) }
  }

  private drainBeforeSeparator(): string {
    const combined = this.tail + this.pending
    const index = combined.indexOf(META_SEPARATOR)
    if (index === -1) {
      // 没有完整分隔符：放行到安全位置，尾部扣住 HOLD_BACK 个字符
      const safeEnd = Math.max(0, combined.length - HOLD_BACK)
      const emit = combined.slice(0, safeEnd)
      this.tail = combined.slice(safeEnd)
      this.pending = ''
      this.bodyEmitted += emit.length
      return emit
    }
    this.separatorSeen = true
    const emit = combined.slice(0, index)
    this.metaRaw = combined.slice(index + META_SEPARATOR.length)
    this.tail = ''
    this.pending = ''
    this.bodyEmitted += emit.length
    return emit
  }
}
```

注意：缺分隔符降级时，`finish()` 返回 null 标题即可；hold-back 尾巴的正文由 `flushTail()` 在流结束时放行（测试 helper 已在循环后调用它），保证正文逐字完整。`flushTail` 实现为：

```ts
  /** 流结束时调用：放行残余 tail（分隔符未出现时它是正文，否则为空）。 */
  flushTail(): string {
    const emit = this.separatorSeen ? '' : this.tail
    this.tail = ''
    this.bodyEmitted += emit.length
    return emit
  }
```

  encodeSseEventV3,
  parseSseStreamV3,
  type SseV3Message,

function streamFrom(parts: string[]): ReadableStream<Uint8Array> {
  const encoder = new TextEncoder()
  return new ReadableStream<Uint8Array>({
    start(controller) {
      for (const part of parts) {
        controller.enqueue(encoder.encode(part))
      }
      controller.close()
    },
  })
}

async function collect(stream: ReadableStream<Uint8Array>): Promise<SseV3Message[]> {
  const out: SseV3Message[] = []

  return out

```ts
import { z } from 'zod'

/** v3 SSE 事件协议（ARCHITECTURE §五）。客户端用 fetch + ReadableStream，不用 EventSource。 */

export const SSE_V3_EVENT_NAMES = ['start', 'delta', 'meta', 'usage', 'error', 'done'] as const
export type SseV3EventName = (typeof SSE_V3_EVENT_NAMES)[number]

export const SSE_V3_HEADERS = {
  'Content-Type': 'text/event-stream; charset=utf-8',
  'Cache-Control': 'no-cache, no-transform',
  'X-Accel-Buffering': 'no',
} as const

export const startPayloadSchema = z.object({
  recordId: z.string().min(1),
  iterationId: z.string().min(1),
})

export const deltaV3PayloadSchema = z.object({
  text: z.string(),
})

export const metaPayloadSchema = z.object({
  title: z.string().nullable(),
  notes: z.array(z.string()),
})

export const usageV3PayloadSchema = z.object({
  tokensIn: z.number().int().nonnegative(),
  tokensOut: z.number().int().nonnegative(),
  costCny: z.number().nonnegative().optional(),
  latencyMs: z.number().nonnegative().optional(),
})

export const errorV3PayloadSchema = z.object({
  code: z.string().min(1),
  message: z.string().min(1),
  retryable: z.boolean(),
})

export const doneV3PayloadSchema = z.object({
  recordId: z.string().min(1),
  iterationId: z.string().min(1),
  status: z.enum(['completed', 'canceled', 'failed']),
})

export type StartPayload = z.infer<typeof startPayloadSchema>
export type DeltaV3Payload = z.infer<typeof deltaV3PayloadSchema>
export type MetaPayload = z.infer<typeof metaPayloadSchema>
export type UsageV3Payload = z.infer<typeof usageV3PayloadSchema>
export type ErrorV3Payload = z.infer<typeof errorV3PayloadSchema>
export type DoneV3Payload = z.infer<typeof doneV3PayloadSchema>

export type SseV3Message =
  | { event: 'start'; data: StartPayload }
  | { event: 'delta'; data: DeltaV3Payload }
  | { event: 'meta'; data: MetaPayload }
  | { event: 'usage'; data: UsageV3Payload }
  | { event: 'error'; data: ErrorV3Payload }
  | { event: 'done'; data: DoneV3Payload }

export function encodeSseEventV3(message: SseV3Message): string {
  return `event: ${message.event}\ndata: ${JSON.stringify(message.data)}\n\n`
}
```

### Task 3: 幂等 LRU（lib/engine/idempotency.ts）

**Files:**
- Create: `lib/engine/idempotency.ts`
- Test: `tests/unit/engine/idempotency.test.ts`

**Interfaces:**
- Consumes: 无。
- Produces（Batch C 三个 SSE 路由消费）:
  - `interface IdempotencyStore<T> { run(requestId: string, executor: () => Promise<T>): Promise<IdempotentOutcome<T>> }`
  - `interface IdempotentOutcome<T> { value: T; replayed: boolean }`
  - `createIdempotencyStore<T>(options?: { ttlMs?: number; capacity?: number; now?: () => number }): IdempotencyStore<T>`（每个测试新建自己的 store，不需要跨测试清空，因此不提供 clear API）

语义：同一 requestId — (1) in-flight 时复用同一个 Promise；(2) 完成后 TTL 内（默认 5 分钟）复用结果；(3) executor 抛错不缓存失败，下一次重新执行；(4) 超容量（默认 500）按最旧完成时间驱逐。`replayed`：首个执行为 false，其余 true。

import { describe, expect, it } from 'vitest'
import { createIdempotencyStore } from '@/lib/engine/idempotency'

describe('幂等 LRU', () => {
  it('并发同 requestId 只执行一次，两个调用拿同一结果', async () => {
    const store = createIdempotencyStore<string>()
    let calls = 0
    const executor = () => {
      calls += 1
      return new Promise<string>((resolve) => setTimeout(() => resolve('done'), 10))
    }
    const [a, b] = await Promise.all([
      store.run('req-1', executor),
      store.run('req-1', executor),
    ])
    expect(calls).toBe(1)
    expect(a.value).toBe('done')
    expect(b.value).toBe('done')
    expect(a.replayed).toBe(false)
    expect(b.replayed).toBe(true)
  })

  it('TTL 内复用完成结果，超过 TTL 重新执行', async () => {
    let now = 0
    const store = createIdempotencyStore<string>({ ttlMs: 5_000, now: () => now })
    let calls = 0
    const executor = async () => {
      calls += 1

```ts
/**
 * 记录层幂等（ARCHITECTURE §十二）。
 *
 * 进程内 LRU：同一 requestId 的 in-flight Promise 直接复用（双击只执行一次），
 * 完成结果在 TTL（默认 5 分钟）内复用。失败不缓存。迭代层由
 * PromptIteration.requestId 数据库唯一约束兜底（Batch C 接入）。
 */

export interface IdempotentOutcome<T> {
  value: T
  replayed: boolean
}

export interface IdempotencyStore<T> {
  run(requestId: string, executor: () => Promise<T>): Promise<IdempotentOutcome<T>>
}

interface Entry<T> {
  promise: Promise<T>
  completedAt: number | null
}

const DEFAULT_TTL_MS = 5 * 60_000
const DEFAULT_CAPACITY = 500

export function createIdempotencyStore<T>(options: {
  ttlMs?: number
  capacity?: number
  now?: () => number
} = {}): IdempotencyStore<T> {
  const ttlMs = options.ttlMs ?? DEFAULT_TTL_MS
  const capacity = options.capacity ?? DEFAULT_CAPACITY
  const now = options.now ?? Date.now
  const entries = new Map<string, Entry<T>>()

  const evictIfNeeded = (): void => {
    if (entries.size <= capacity) {
      return
    }
    // 最旧完成时间优先；未完成的排最后（不允许驱逐 in-flight）
    let oldestKey: string | null = null
    let oldestAt = Number.POSITIVE_INFINITY
    for (const [key, entry] of entries) {
      const at = entry.completedAt ?? Number.POSITIVE_INFINITY
      if (at < oldestAt) {
        oldestAt = at
        oldestKey = key
      }
    }
    if (oldestKey !== null && entries.get(oldestKey)?.completedAt !== null) {
      entries.delete(oldestKey)
    }
  }

  return {
    run(requestId, executor) {
      const existing = entries.get(requestId)
      if (existing) {
        if (existing.completedAt === null || now() - existing.completedAt < ttlMs) {
          return existing.promise.then((value) => ({ value, replayed: true }))
        }
        entries.delete(requestId)
      }

      const entry: Entry<T> = { promise: executor(), completedAt: null }
      entries.set(requestId, entry)
      entry.promise.then(
        () => {
          entry.completedAt = now()
          evictIfNeeded()
        },
        () => {
          // 失败不缓存：移除，让下一次调用重新执行
          entries.delete(requestId)
        },
      )
      return entry.promise.then((value) => ({ value, replayed: false }))
    },
  }
}
```

实现说明：不需要 clear API，每个测试新建自己的 store；测试文件顶部不需要 `vi` 导入（用注入 `now` 时钟替代真实时间控制）。
